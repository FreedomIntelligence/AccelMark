# AccelMark -- Ascend NPU vllm-ascend runner dependencies
#
# This runner is designed to run inside the official vllm-ascend Docker image.
# All heavy dependencies (torch, torch_npu, vllm, vllm-ascend, CANN) are
# pre-installed in the image -- do NOT reinstall them via pip.
#
# Tested image: quay.io/ascend/vllm-ascend:v0.18.0rc1-a3
# Setup guide:  https://docs.vllm.ai/projects/ascend/en/latest/installation.html#set-up-using-docker
#
# ── Method 1: docker run with bind mounts (works on most systems) ──────────
#
# Example docker run (update --device flags to match your hardware):
#   # Atlas A2: /dev/davinci[0-7]   Atlas A3: /dev/davinci[0-15]
#   export IMAGE=quay.io/ascend/vllm-ascend:v0.18.0rc1-a3
#   docker run --rm \
#       --name vllm-ascend-env \
#       --privileged \
#       --shm-size=1g \
#       --net=host \
#       --device /dev/davinci0 \
#       --device /dev/davinci1 \
#       --device /dev/davinci2 \
#       --device /dev/davinci3 \
#       --device /dev/davinci4 \
#       --device /dev/davinci5 \
#       --device /dev/davinci6 \
#       --device /dev/davinci7 \
#       --device /dev/davinci_manager \
#       --device /dev/devmm_svm \
#       --device /dev/hisi_hdc \
#       -v /usr/local/dcmi:/usr/local/dcmi \
#       -v $(which npu-smi):/usr/bin/npu-smi \
#       -v /usr/local/Ascend:/usr/local/Ascend \
#       -v /usr/local/Ascend/driver/lib64/:/usr/local/Ascend/driver/lib64/ \
#       -v /usr/local/Ascend/driver/version.info:/usr/local/Ascend/driver/version.info \
#       -v /etc/ascend_install.info:/etc/ascend_install.info \
#       -v /root/.cache:/root/.cache \
#       -it $IMAGE bash
#
#   NOTE: --privileged may be required on some systems. If npu-smi returns
#   error -9005 (get board type failed), adding --privileged usually fixes it.
#   collect_env.py depends on npu-smi to enumerate accelerators; without it,
#   accelerators is empty → benchmark_runner.py exits with an IndexError hint.
#
#   The npu-smi mount uses $(which npu-smi) to dynamically resolve the host
#   path. If that doesn't work, replace it with the actual host path found by
#   `which npu-smi` (common: /usr/local/sbin/npu-smi or /usr/local/bin/npu-smi).
#
# ── Method 2: docker cp (fallback for containerd / overlayfs environments) ──
#
#   On some systems (e.g. containerd with overlayfs snapshotters), Docker's
#   `-v /host/path:/container/path` bind mount is silently converted to an
#   overlay mount, resulting in an empty directory inside the container
#   instead of the expected file. The Ascend driver directories also fail
#   to mount. In that case, copy files in after starting the container:
#
#   # Step 1: Start a background container
#   docker run -d \
#       --name vllm-ascend-env \
#       --shm-size=1g \
#       --net=host \
#       --privileged \
#       --device /dev/davinci0 \
#       ... (all --device flags as in Method 1) ...
#       --entrypoint sleep \
#       $IMAGE infinity
#
#   # Step 2: Copy npu-smi binary into the container
#   docker cp $(which npu-smi) vllm-ascend-env:/usr/bin/npu-smi
#
#   # Step 3: Copy Ascend driver shared libraries
#   #   Find your CANN version first:  ls /usr/local/Ascend/  (e.g. cann-8.5.0)
#   #   Then copy the required .so files (adjust CANN_VERSION to match):
#   CANN_VER=$(ls /usr/local/Ascend/ | grep cann- | head -1)
#   docker cp /usr/local/Ascend/$CANN_VER/aarch64-linux/lib64/libc_sec.so \
#       vllm-ascend-env:/lib/aarch64-linux-gnu/
#   docker cp /usr/local/Ascend/$CANN_VER/aarch64-linux/lib64/libmmpa.so \
#       vllm-ascend-env:/lib/aarch64-linux-gnu/
#   docker cp /usr/local/Ascend/driver/lib64/driver/libdrvdsmi_host.so \
#       vllm-ascend-env:/lib/aarch64-linux-gnu/
#   docker cp /usr/local/Ascend/driver/lib64/driver/libascend_hal.so \
#       vllm-ascend-env:/lib/aarch64-linux-gnu/
#
#   # Step 4: Refresh linker cache
#   docker exec vllm-ascend-env ldconfig
#
#   # Step 5: Verify
#   docker exec -it vllm-ascend-env npu-smi info
#
#   # Step 6: Enter interactively and install AccelMark
#   docker exec -it vllm-ascend-env /bin/bash
#
# Inside the container, install only the AccelMark-specific extras:
#   pip install -r runners/ascend_vllm_ascend_d4aa9fda/requirements.txt
#
# Pre-installed in image (versions from v0.18.0rc1-a3):
